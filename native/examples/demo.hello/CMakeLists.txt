cmake_minimum_required(VERSION 3.22)
project(beacon_hello LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
  message(FATAL_ERROR "Use a Windows x86 toolchain")
endif()
set(BEACON_SDK_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." CACHE PATH "Beacon native SDK root")
add_library(plugin SHARED plugin.cpp)
target_compile_features(plugin PRIVATE cxx_std_17)
target_include_directories(plugin PRIVATE "${BEACON_SDK_ROOT}/include" "${BEACON_SDK_ROOT}/vendor")
set_target_properties(plugin PROPERTIES PREFIX "" OUTPUT_NAME "plugin")
if(MINGW)
  target_compile_options(plugin PRIVATE -O2 -g0 -Werror=date-time)
  target_link_options(plugin PRIVATE -static -s -Wl,--no-insert-timestamp)
endif()
configure_file(manifest.json manifest.json COPYONLY)
