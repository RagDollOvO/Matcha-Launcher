function(l4p_target target)
  target_include_directories(${target} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/include" "${CMAKE_CURRENT_SOURCE_DIR}/src" "${CMAKE_CURRENT_SOURCE_DIR}/vendor")
  if(target STREQUAL "skeeto_crosshair")
    target_compile_definitions(${target} PRIVATE WINVER=0x0A00 _WIN32_WINNT=0x0A00 UNICODE _UNICODE)
  else()
    target_compile_definitions(${target} PRIVATE WINVER=0x0602 _WIN32_WINNT=0x0602 UNICODE _UNICODE)
  endif()
  if(MINGW)
    target_link_options(${target} PRIVATE -static)
  endif()
endfunction()

function(export_file file prefix marker)
  file(WRITE "${file}" "EXPORTS\n")
  foreach(item IN LISTS EXPORTS)
    string(REPLACE "," ";" parts "${item}")
    list(GET parts 0 name)
    list(GET parts 1 ordinal)
    list(GET parts 2 bytes)
    list(GET parts 3 flags)
    set(symbol "${prefix}_${name}")
    if(CMAKE_SIZEOF_VOID_P EQUAL 4)
      string(APPEND symbol "@${bytes}")
      if(MSVC)
        set(symbol "_${symbol}")
      endif()
    endif()
    file(APPEND "${file}" "  ${name}=${symbol} @${ordinal} ${flags}\n")
  endforeach()
  if(marker)
    file(APPEND "${file}" "  L4P_PlatformShimVersion @100\n")
    file(APPEND "${file}" "  L4P_RenderBackendControlV1 @101\n")
  endif()
endfunction()

function(source_plugin target id)
  add_library(${target} SHARED "plugins/${id}/plugin.cpp")
  l4p_target(${target})
  if(MSVC)
    target_compile_options(${target} PRIVATE /utf-8)
  endif()
  set(output "${DIST}/bin/l4p/plugins/${id}")
  set_target_properties(${target} PROPERTIES PREFIX "" OUTPUT_NAME "plugin" RUNTIME_OUTPUT_DIRECTORY "${output}" ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/${target}")
  foreach(name manifest.json LICENSE SOURCE.md)
    configure_file("plugins/${id}/${name}" "${output}/${name}" COPYONLY)
  endforeach()
  if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/plugins/${id}/catalog.png")
    configure_file("plugins/${id}/catalog.png" "${output}/catalog.png" COPYONLY)
  endif()
  configure_file("plugins/${id}/plugin.cpp" "${output}/source/plugin.cpp" COPYONLY)
  configure_file("plugins/${id}/Standalone.cmake" "${output}/source/CMakeLists.txt" COPYONLY)
  foreach(name l4p_plugin.h l4p_source.h l4p_cvar.h l4p_sound.h l4p_entity.h l4p_effects.h l4p_audio.h l4p_resource.h l4p_draw.h l4p_view.h l4p_addons.h l4p_model.h l4p_file.h l4p_glow.h l4p_input.h l4p_weapon.h l4p_player.h l4p_localize.h l4p_corpse.h l4p_viewmodel.h l4p_dissolve.h l4p_particle.h l4p_feedback.h)
    configure_file("include/${name}" "${output}/source/${name}" COPYONLY)
  endforeach()
  configure_file(vendor/nlohmann/json.hpp "${output}/source/nlohmann/json.hpp" COPYONLY)
  if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/plugins/${id}/preset.json")
    target_include_directories(${target} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/plugins/shared")
    foreach(name preset.json feature.hpp)
      configure_file("plugins/${id}/${name}" "${output}/source/${name}" COPYONLY)
    endforeach()
    configure_file(plugins/shared/preset_plugin.inc "${output}/source/preset_plugin.inc" COPYONLY)
  endif()
endfunction()
